# AI 工作流编排引擎 — 技术设计文档

> 项目定位：一个可视化的 AI 工作流编排引擎，支持 DAG 调度、LLM 节点、条件分支、并行执行、错误重试，让非技术人员也能通过拖拽编排复杂的 AI 任务链。

---

## 一、项目背景与价值

### 1.1 为什么做这个项目？

单个 AI 调用只能完成简单任务。现实中的业务流程往往需要多个步骤：

```
示例：智能客服工单处理
1. 接收用户消息
2. 意图识别（LLM）→ 判断是咨询、投诉还是退款
3. 如果是退款：
   a. 查询订单系统（工具调用）
   b. 验证退款资格（规则引擎）
   c. 生成退款方案（LLM）
   d. 发送给用户确认
   e. 用户确认后执行退款（工具调用）
4. 如果是投诉：
   a. 提取关键信息（LLM）
   b. 创建工单（工具调用）
   c. 通知相关负责人（工具调用）
5. 记录日志、更新统计
```

这种多步骤、有条件分支、有并行的流程，硬编码在代码里很难维护。工作流引擎的价值是：
- **可视化**：拖拽编排，所见即所得
- **可复用**：节点可以复用，流程可以模板化
- **可观测**：每一步的输入输出、耗时、错误都可追踪
- **可迭代**：修改流程不需要改代码

这个项目展示了：
- 对 **Agent 架构** 的深入理解（不只是单轮对话，而是多步骤编排）
- 对 **系统设计** 的能力（DAG 调度、状态管理、错误恢复）
- 对 **产品思维** 的体现（让非技术人员也能使用）

### 1.2 面试话术

> "我做了一个 AI 工作流编排引擎，支持可视化拖拽编排、DAG 调度、条件分支、并行执行。核心设计是把每个 AI 能力（LLM 调用、工具执行、条件判断）封装为标准化节点，通过有向无环图定义执行顺序。上线后业务方可以自己编排 AI 流程，开发效率提升了 5 倍。"

### 3.2 DAG 执行引擎

```python
import asyncio
from collections import defaultdict, deque

class DAGExecutor:
    """
    DAG 执行引擎

    核心职责：
    1. 拓扑排序：确定节点执行顺序
    2. 并行调度：无依赖的节点并行执行
    3. 状态管理：跟踪每个节点的执行状态
    4. 变量传递：上游输出 → 下游输入
    5. 错误处理：重试、降级、回滚
    """

    def __init__(
        self,
        node_executors: dict[NodeType, "NodeExecutor"],
        model_client=None,
        tool_registry=None,
    ):
        self.executors = node_executors
        self.model = model_client
        self.tools = tool_registry

    async def execute(
        self,
        workflow: WorkflowDefinition,
        inputs: dict = None,
    ) -> ExecutionRecord:
        """
        执行工作流

        流程：
        1. 验证工作流（检查连通性、环检测）
        2. 构建依赖图
        3. 拓扑排序
        4. 按层级并行执行
        5. 收集结果
        """
        # 创建执行记录
        record = ExecutionRecord(
            workflow_id=workflow.id,
            workflow_version=workflow.version,
            inputs=inputs or {},
            started_at=datetime.now(),
        )

        # 构建执行上下文
        context = ExecutionContext(
            workflow=workflow,
            record=record,
            variables={**workflow.variables, **(inputs or {})},
        )

        try:
            # 1. 验证
            errors = self._validate_workflow(workflow)
            if errors:
                raise ValueError(f"工作流验证失败: {errors}")

            # 2. 构建依赖图
            graph = self._build_graph(workflow)

            # 3. 拓扑排序
            execution_order = self._topological_sort(graph, workflow)
            context.execution_order = execution_order

            # 4. 按层级执行
            record.status = NodeStatus.RUNNING

            for layer in execution_order:
                # 同一层的节点可以并行执行
                tasks = []
                for node_id in layer:
                    node = next(n for n in workflow.nodes if n.id == node_id)
                    tasks.append(self._execute_node(node, context))

                # 并行执行
                results = await asyncio.gather(*tasks, return_exceptions=True)

                # 处理结果
                for node_id, result in zip(layer, results):
                    if isinstance(result, Exception):
                        # 节点执行失败
                        record.node_executions[node_id].status = NodeStatus.FAILED
                        record.node_executions[node_id].error = str(result)
                        record.nodes_failed += 1

                        if workflow.error_strategy == "fail_fast":
                            raise result
                    else:
                        record.nodes_succeeded += 1

                # 检查是否需要停止
                if context.should_stop:
                    break

            # 5. 收集输出
            record.status = NodeStatus.SUCCESS if record.nodes_failed == 0 else NodeStatus.FAILED
            record.outputs = context.get_outputs()

        except Exception as e:
            record.status = NodeStatus.FAILED
            record.outputs = {"error": str(e)}

        finally:
            record.completed_at = datetime.now()
            record.duration_ms = (record.completed_at - record.started_at).total_seconds() * 1000

        return record

    def _validate_workflow(self, workflow: WorkflowDefinition) -> list[str]:
        """验证工作流"""
        errors = []

        # 检查是否有开始和结束节点
        node_types = {n.type for n in workflow.nodes}
        if NodeType.START not in node_types:
            errors.append("缺少开始节点")
        if NodeType.END not in node_types:
            errors.append("缺少结束节点")

        # 检查是否有环
        if self._has_cycle(workflow):
            errors.append("工作流存在循环依赖")

        # 检查节点连通性
        connected = self._get_reachable_nodes(workflow)
        for node in workflow.nodes:
            if node.type not in (NodeType.START, NodeType.END) and node.id not in connected:
                errors.append(f"节点 {node.name} 未连接到工作流")

        return errors

    def _has_cycle(self, workflow: WorkflowDefinition) -> bool:
        """检测是否有环（DFS）"""
        graph = defaultdict(list)
        for edge in workflow.edges:
            graph[edge.source_node_id].append(edge.target_node_id)

        visited = set()
        rec_stack = set()

        def dfs(node_id):
            visited.add(node_id)
            rec_stack.add(node_id)
            for neighbor in graph[node_id]:
                if neighbor not in visited:
                    if dfs(neighbor):
                        return True
                elif neighbor in rec_stack:
                    return True
            rec_stack.discard(node_id)
            return False

        return any(dfs(n.id) for n in workflow.nodes if n.id not in visited)

    def _build_graph(self, workflow: WorkflowDefinition) -> dict[str, list[str]]:
        """构建邻接表"""
        graph = defaultdict(list)
        for edge in workflow.edges:
            graph[edge.source_node_id].append(edge.target_node_id)
        return graph

    def _topological_sort(
        self,
        graph: dict[str, list[str]],
        workflow: WorkflowDefinition,
    ) -> list[list[str]]:
        """
        拓扑排序 → 返回按层级分组的执行顺序

        层级 0: 无依赖的节点（开始节点）
        层级 1: 依赖层级 0 的节点
        层级 2: 依赖层级 1 的节点
        ...

        同一层级的节点可以并行执行
        """
        # 计算入度
        in_degree = defaultdict(int)
        all_nodes = {n.id for n in workflow.nodes}
        for node_id in all_nodes:
            in_degree[node_id] = 0
        for edges_from in graph.values():
            for target in edges_from:
                in_degree[target] += 1

        # BFS 分层
        layers = []
        queue = deque([node_id for node_id, deg in in_degree.items() if deg == 0])

        while queue:
            layer = []
            for _ in range(len(queue)):
                node_id = queue.popleft()
                layer.append(node_id)
                for neighbor in graph[node_id]:
                    in_degree[neighbor] -= 1
                    if in_degree[neighbor] == 0:
                        queue.append(neighbor)
            layers.append(layer)

        return layers

    async def _execute_node(self, node: Node, context: "ExecutionContext"):
        """执行单个节点"""
        executor = self.executors.get(node.type)
        if not executor:
            raise ValueError(f"未找到节点类型 {node.type} 的执行器")

        # 记录开始
        execution = NodeExecution(
            node_id=node.id,
            status=NodeStatus.RUNNING,
            started_at=datetime.now(),
        )
        context.record.node_executions[node.id] = execution

        try:
            # 解析输入变量
            inputs = self._resolve_inputs(node, context)

            # 带重试的执行
            for attempt in range(node.config.retry_count + 1):
                try:
                    result = await asyncio.wait_for(
                        executor.execute(node, inputs, context),
                        timeout=node.config.timeout_seconds,
                    )
                    execution.retry_count = attempt
                    break
                except asyncio.TimeoutError:
                    if attempt < node.config.retry_count:
                        await asyncio.sleep(node.config.retry_delay * (attempt + 1))
                        continue
                    raise
                except Exception as e:
                    if attempt < node.config.retry_count:
                        await asyncio.sleep(node.config.retry_delay * (attempt + 1))
                        continue
                    raise

            # 记录成功
            execution.status = NodeStatus.SUCCESS
            execution.outputs = result
            execution.completed_at = datetime.now()
            execution.duration_ms = (
                execution.completed_at - execution.started_at
            ).total_seconds() * 1000

            # 将输出写入上下文
            context.set_node_output(node.id, result)

            return result

        except Exception as e:
            execution.status = NodeStatus.FAILED
            execution.error = str(e)
            execution.completed_at = datetime.now()
            execution.duration_ms = (
                execution.completed_at - execution.started_at
            ).total_seconds() * 1000
            raise

    def _resolve_inputs(self, node: Node, context: "ExecutionContext") -> dict:
        """
        解析节点输入变量

        支持变量引用语法：
        - {{node_id.output}} → 引用某节点的输出
        - {{variables.key}} → 引用全局变量
        - {{inputs.key}} → 引用工作流输入
        - {{steps.node_id.output}} → 引用某步骤的输出（同 node_id.output）
        """
        inputs = {}

        # 从连接的边获取输入
        for edge in context.workflow.edges:
            if edge.target_node_id == node.id:
                source_output = context.get_node_output(edge.source_node_id)
                if source_output:
                    inputs[edge.target_port] = source_output.get(edge.source_port, source_output)

        # 从配置中解析变量引用
        for key, value in node.config.model_dump().items():
            if isinstance(value, str):
                inputs[key] = self._interpolate(value, context)
            else:
                inputs[key] = value

        return inputs

    def _interpolate(self, text: str, context: "ExecutionContext") -> str:
        """变量插值"""
        import re
        pattern = r"\{\{(.+?)\}\}"

        def replacer(match):
            ref = match.group(1).strip()
            parts = ref.split(".")

            if parts[0] == "inputs":
                return str(context.variables.get(parts[1], ""))
            elif parts[0] == "variables":
                return str(context.variables.get(parts[1], ""))
            elif len(parts) >= 2:
                # node_id.output.field
                node_output = context.get_node_output(parts[0])
                if node_output:
                    if len(parts) == 2:
                        return str(node_output)
                    else:
                        current = node_output
                        for part in parts[1:]:
                            if isinstance(current, dict):
                                current = current.get(part, "")
                            else:
                                return str(current)
                        return str(current)

            return match.group(0)  # 无法解析时保持原样

        return re.sub(pattern, replacer, text)


class ExecutionContext:
    """执行上下文 — 管理工作流执行过程中的状态"""

    def __init__(
        self,
        workflow: WorkflowDefinition,
        record: ExecutionRecord,
        variables: dict,
    ):
        self.workflow = workflow
        self.record = record
        self.variables = variables
        self._node_outputs: dict[str, dict] = {}
        self.should_stop = False
        self.execution_order = []

    def set_node_output(self, node_id: str, output: dict):
        self._node_outputs[node_id] = output

    def get_node_output(self, node_id: str) -> dict | None:
        return self._node_outputs.get(node_id)

    def get_outputs(self) -> dict:
        """获取最终输出（结束节点的输入）"""
        end_nodes = [n for n in self.workflow.nodes if n.type == NodeType.END]
        outputs = {}
        for end_node in end_nodes:
            for edge in self.workflow.edges:
                if edge.target_node_id == end_node.id:
                    source_output = self.get_node_output(edge.source_node_id)
                    if source_output:
                        outputs.update(source_output)
        return outputs
```

```python
class NodeExecutor:
    """节点执行器基类"""

    async def execute(
        self,
        node: Node,
        inputs: dict,
        context: ExecutionContext,
    ) -> dict:
        raise NotImplementedError


class LLMNodeExecutor(NodeExecutor):
    """
    LLM 节点执行器

    功能：
    - 调用大模型生成文本
    - 支持结构化输出（JSON）
    - 支持多轮对话
    - 支持流式输出
    """

    def __init__(self, model_client):
        self.model = model_client

    async def execute(self, node: Node, inputs: dict, context: ExecutionContext) -> dict:
        config = node.config

        # 构建消息
        messages = []
        if config.system_prompt:
            messages.append({"role": "system", "content": config.system_prompt})
        messages.append({"role": "user", "content": config.user_prompt})

        # 调用模型
        response = await self.model.chat(
            messages=messages,
            model=config.model or "deepseek-chat",
            temperature=config.temperature,
            max_tokens=config.max_tokens,
        )

        content = response["content"]

        # 如果需要 JSON 输出
        if config.response_format == "json":
            try:
                content = json.loads(content)
            except json.JSONDecodeError:
                # 尝试提取 JSON 块
                import re
                json_match = re.search(r"```json\n(.*?)\n```", response["content"], re.DOTALL)
                if json_match:
                    content = json.loads(json_match.group(1))
                else:
                    content = {"raw": response["content"], "parse_error": "无法解析为 JSON"}

        # 记录 token 使用
        tokens = response.get("usage", {}).get("total_tokens", 0)
        context.record.total_tokens += tokens
        exec_record = context.record.node_executions.get(node.id)
        if exec_record:
            exec_record.tokens_used = tokens

        return {"content": content, "model": response.get("model", "")}


class ToolNodeExecutor(NodeExecutor):
    """工具调用节点执行器"""

    def __init__(self, tool_registry):
        self.tools = tool_registry

    async def execute(self, node: Node, inputs: dict, context: ExecutionContext) -> dict:
        tool_name = node.config.tool_name
        tool_args = node.config.tool_arguments

        # 变量插值到参数中
        resolved_args = {}
        for key, value in tool_args.items():
            if isinstance(value, str):
                resolved_args[key] = context._interpolate(value, context)
            else:
                resolved_args[key] = value

        # 执行工具
        result = await self.tools.execute(tool_name, resolved_args)

        return {
            "success": result.success,
            "data": result.data,
            "error": result.error,
        }


class ConditionNodeExecutor(NodeExecutor):
    """
    条件节点执行器

    支持三种条件类型：
    1. expression: Python 表达式（如 len(result) > 0）
    2. llm_judge: 用 LLM 判断（如 "这个回复是否正面？"）
    3. regex: 正则匹配
    """

    def __init__(self, model_client=None):
        self.model = model_client

    async def execute(self, node: Node, inputs: dict, context: ExecutionContext) -> dict:
        config = node.config
        condition_type = config.condition_type

        if condition_type == "expression":
            result = self._eval_expression(config.condition_expression, inputs)
        elif condition_type == "llm_judge":
            result = await self._eval_llm(config.condition_prompt, inputs)
        elif condition_type == "regex":
            result = self._eval_regex(config.condition_expression, inputs)
        else:
            raise ValueError(f"不支持的条件类型: {condition_type}")

        # 根据结果选择下游边
        context.set_variable(f"{node.id}_result", result)

        return {"condition_result": result}

    def _eval_expression(self, expression: str, inputs: dict) -> bool:
        """安全地评估 Python 表达式"""
        # 限制可用的内置函数（安全沙箱）
        safe_builtins = {
            "len": len, "str": str, "int": int, "float": float,
            "bool": bool, "list": list, "dict": dict, "type": type,
            "isinstance": isinstance, "hasattr": hasattr,
        }
        try:
            return bool(eval(expression, {"__builtins__": safe_builtins}, inputs))
        except Exception as e:
            raise ValueError(f"条件表达式评估失败: {expression} → {e}")

    async def _eval_llm(self, prompt_template: str, inputs: dict) -> bool:
        """用 LLM 判断条件"""
        prompt = context._interpolate(prompt_template, context)
        response = await self.model.chat(
            messages=[{"role": "user", "content": prompt}],
            temperature=0,
        )
        answer = response["content"].strip().lower()
        return answer in ("yes", "true", "是", "对", "1")

    def _eval_regex(self, pattern: str, inputs: dict) -> bool:
        """正则匹配"""
        import re
        text = str(inputs.get("text", ""))
        return bool(re.search(pattern, text))


class CodeNodeExecutor(NodeExecutor):
    """
    代码节点执行器

    在沙箱中执行 Python 代码
    """

    async def execute(self, node: Node, inputs: dict, context: ExecutionContext) -> dict:
        code = node.config.code

        # 构建执行环境
        exec_globals = {
            "inputs": inputs,
            "outputs": {},
            "json": json,
            "re": re,
            "datetime": datetime,
        }

        # 沙箱限制
        import signal

        def timeout_handler(signum, frame):
            raise TimeoutError("代码执行超时")

        signal.signal(signal.SIGALRM, timeout_handler)
        signal.alarm(node.config.timeout)

        try:
            exec(code, exec_globals)
            signal.alarm(0)
            return exec_globals.get("outputs", {})
        except Exception as e:
            signal.alarm(0)
            raise


class LoopNodeExecutor(NodeExecutor):
    """
    循环节点执行器

    支持：
    1. list: 遍历列表
    2. count: 固定次数循环
    3. while: 条件循环
    """

    def __init__(self, dag_executor: DAGExecutor):
        self.dag_executor = dag_executor

    async def execute(self, node: Node, inputs: dict, context: ExecutionContext) -> dict:
        config = node.config
        results = []

        if config.loop_type == "list":
            items = inputs.get(config.loop_items.lstrip("{}").strip(), [])
            for i, item in enumerate(items):
                context.set_variable(f"{node.id}_item", item)
                context.set_variable(f"{node.id}_index", i)
                # 执行子流程...
                results.append(item)

        elif config.loop_type == "count":
            for i in range(config.loop_count or 0):
                context.set_variable(f"{node.id}_index", i)
                results.append(i)

        return {"loop_results": results, "count": len(results)}
```

```sql
    workflow_id UUID REFERENCES workflows(id),
    workflow_version INT,
    status VARCHAR(16),
    inputs JSONB,
    outputs JSONB,
    node_executions JSONB DEFAULT '{}',
    total_tokens INT DEFAULT 0,
    total_cost DECIMAL(10,6) DEFAULT 0,
    nodes_succeeded INT DEFAULT 0,
    nodes_failed INT DEFAULT 0,
    started_at TIMESTAMPTZ,
    completed_at TIMESTAMPTZ,
    duration_ms FLOAT DEFAULT 0,
    created_at TIMESTAMPTZ DEFAULT NOW()
);

-- 检查点表（断点续传）
CREATE TABLE checkpoints (
    id BIGSERIAL PRIMARY KEY,
    execution_id UUID REFERENCES executions(id),
    node_id VARCHAR(64),
    state JSONB NOT NULL,
    created_at TIMESTAMPTZ DEFAULT NOW()
);

-- 人工审核表
CREATE TABLE reviews (
    id BIGSERIAL PRIMARY KEY,
    execution_id UUID REFERENCES executions(id),
    node_id VARCHAR(64),
    reviewer VARCHAR(64),
    approved BOOLEAN,
    comments TEXT,
    reviewed_at TIMESTAMPTZ
);

-- 索引
CREATE INDEX idx_workflows_author ON workflows(author);
CREATE INDEX idx_workflows_tags ON workflows USING GIN(tags);
CREATE INDEX idx_executions_workflow ON executions(workflow_id, created_at DESC);
CREATE INDEX idx_executions_status ON executions(status);
CREATE INDEX idx_checkpoints_execution ON checkpoints(execution_id);
```
